02912: /** 构造某一步 DP 的候选切点 */

02960: /** 用 DP 规划在 targetChunkCount 段下的最佳切法 */

02963:   const totalUnits = units.length;
02964:   const totalSize = prefix[totalUnits];

02970:     if (memo.has(key)) return memo.get(key);

02973:     const remainingSize = totalSize - prefix[start];
02974: 
02975:     if (remainingSize > remainingChunkCount * HARD_CHUNK_LIMIT) {
02976:       memo.set(key, null);
02977:       return null;
02978:     }
02979: 
02980:     if (chunkIndex === targetChunkCount - 1) {

02983:         memo.set(key, null);
02984:         return null;
02985:       }
02986: 
02987:       const result = {
02988:         score: scoreChunkSize(finalSize, idealSize),
02989:         ends: [totalUnits],
02990:       };
02991:       memo.set(key, result);
02992:       return result;
02993:     }
02994: 
02995:     const candidateEnds = buildCandidateEnds({
02996:       units,
02997:       prefix,
02998:       start,
02999:       chunkIndex,
03000:       targetChunkCount,
03001:     });

03004: 
03005:     for (const end of candidateEnds) {

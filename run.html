02491: function stabilizePolishMods(mods = []) {
02492:   const candidates = (mods || [])
02493:     .filter(
02494:       (mod) =>
02495:         Number.isInteger(mod.absoluteStart) &&
02496:         Number.isInteger(mod.absoluteEnd) &&
02497:         typeof mod.old_text === 'string' &&
02498:         typeof mod.new_text === 'string' &&
02499:         mod.absoluteEnd >= mod.absoluteStart &&
02500:         mod.old_text,
02501:     )
02502:     .map((mod, idx) => ({
02503:       ...mod,
02504:       __priority: getPolishModPriority(mod),
02505:       __length: Math.max(0, mod.absoluteEnd - mod.absoluteStart),
02506:       __index: idx,
02507:     }))
02508:     .sort((a, b) => {
02509:       if (b.__priority !== a.__priority) return b.__priority - a.__priority;
02510:       if (b.__length !== a.__length) return b.__length - a.__length;
02511:       if (a.absoluteStart !== b.absoluteStart) return a.absoluteStart - b.absoluteStart;
02512:       return a.__index - b.__index;
02513:     });
02514: 
02515:   const accepted = [];
02516: 
02517:   function overlaps(existing, current) {
02518:     return current.absoluteStart < existing.absoluteEnd && current.absoluteEnd > existing.absoluteStart;
02519:   }
02520: 
02521:   for (const mod of candidates) {
02522:     if (accepted.some((x) => overlaps(x, mod))) continue;
02523:     accepted.push(mod);
02524:   }
02525: 
02526:   return accepted
02527:     .sort((a, b) => {
02528:       if (a.absoluteStart !== b.absoluteStart) return a.absoluteStart - b.absoluteStart;
02529:       if (a.absoluteEnd !== b.absoluteEnd) return a.absoluteEnd - b.absoluteEnd;
02530:       return a.__index - b.__index;

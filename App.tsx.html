517function splitTextIntoEvenChunks(text: string, chunkSize: number): ChunkInfo[] {
518 const s = String(text || "");
519 if (!s) return [];
520 const maxChunk = Math.max(800, Number(chunkSize || DEFAULT_CHUNK_SIZE));
521 const totalLength = s.length;
522 if (totalLength <= maxChunk) return [{ text: s, startIndex: 0, endIndex: totalLength }];
523
524 const minChunks = Math.ceil(totalLength / maxChunk);
525 const targetSize = Math.ceil(totalLength / minChunks);
526 const chunks: ChunkInfo[] = [];
527 let cursor = 0;
528
529 const isBoundaryAt = (idx: number) => {
530 const prev = s[idx - 1] || "";
531 const next = s[idx] || "";
532 if (prev === "\n" && next === "\n") return true;
533 if (/[。！？!?；;]/.test(prev)) return true;
534 if (prev === "\n") return true;
535 return false;
536 };
537
538 for (let i = 0; i < minChunks; i += 1) {
539 if (cursor >= totalLength) break;
540 if (i === minChunks - 1) {
541 chunks.push({ text: s.slice(cursor), startIndex: cursor, endIndex: totalLength });
542 break;
543 }
544 const windowStart = cursor + Math.floor(targetSize * 0.7);
545 const windowEnd = Math.min(cursor + Math.floor(targetSize * 1.15), totalLength - 1);
546
547 let endIndex = -1;
548 for (let j = Math.min(cursor + targetSize, windowEnd); j >= windowStart; j -= 1) {
549 if (isBoundaryAt(j)) {
550 endIndex = j;
551 break;
552 }
553 }
554 if (endIndex === -1) {
555 for (let j = Math.min(cursor + targetSize, windowEnd); j <= windowEnd; j += 1) {
556 if (isBoundaryAt(j)) {
557 endIndex = j;
558 break;
559 }
560 }
561 }
562 if (endIndex === -1 || endIndex <= cursor) endIndex = Math.min(cursor + targetSize, totalLength);
563 chunks.push({ text: s.slice(cursor, endIndex), startIndex: cursor, endIndex });
564 cursor = endIndex;
565 }
566
567 return chunks.filter((c) => c.text.length > 0);
568}

572function buildSentenceChunks(text: string): ChunkInfo[] {
573 const s = String(text || "");
574 if (!s) return [];

577 let i = 0;
578
579 while (i < s.length) {
580 const ch = s[i];
581
582 if (ch === "\n" && s[i + 1] === "\n") {
583 let end = i + 2;

586 chunks.push({ text: s.slice(start, end), startIndex: start, endIndex: end });
587 }
588 start = end;
589 i = end;
590 continue;
591 }
592
593 if (/[。！？!?；;]/.test(ch)) {
594 let end = i + 1;
595 while (end < s.length && SENTENCE_CLOSERS.has(s[end])) end += 1;
596 while (end < s.length && s[end] === " ") end += 1;
597 if (end < s.length && s[end] === "\n" && s[end + 1] !== "\n") end += 1;

609 chunks.push({ text: s.slice(start, end), startIndex: start, endIndex: end });
610 }
611 start = end;
612 i = end;
613 continue;
614 }
615
616 i += 1;
617 }
618
619 if (start < s.length) {
620 chunks.push({ text: s.slice(start), startIndex: start, endIndex: s.length });
621 }

623 return chunks.filter((item) => item.text.length > 0);

642function splitChunkIntoBalancedSentenceParts(text: string, partCount: number): ChunkInfo[] {
643 const s = String(text || "");
644 const totalLength = s.length;
645 const targetParts = Math.max(1, Math.floor(partCount));
646 if (!s || totalLength === 0) return [];
647 if (targetParts <= 1) return [{ text: s, startIndex: 0, endIndex: totalLength }];
648
649 const sentenceChunks = buildSentenceChunks(s);
650 if (sentenceChunks.length <= 1) return splitChunkIntoFixedParts(s, targetParts);
651
652 const actualParts = Math.min(targetParts, sentenceChunks.length);
653 if (actualParts <= 1) return [{ text: s, startIndex: 0, endIndex: totalLength }];
654
655 const prefix: number[] = [0];
656 for (const item of sentenceChunks) {
657 prefix.push(prefix[prefix.length - 1] + item.text.length);
658 }
659
660 const sentenceCount = sentenceChunks.length;
661 const ideal = totalLength / actualParts;
662 const minAllowed = ideal * 0.8;
663 const maxAllowed = ideal * 1.2;
664
665 let bestRanges: Array<[number, number]> | null = null;
666 let bestOverflow = Number.POSITIVE_INFINITY;
667 let bestPenalty = Number.POSITIVE_INFINITY;
668
669 function partScore(startSentence: number, endSentenceExclusive: number) {
670 const len = prefix[endSentenceExclusive] - prefix[startSentence];
671 const overflow =
672 len < minAllowed
673 ? minAllowed - len
674 : len > maxAllowed
675 ? len - maxAllowed
676 : 0;
677 const deviation = Math.abs(len - ideal);
678 return { len, overflow, deviation };
679 }
680
681 function dfs(
682 partIndex: number,
683 startSentence: number,
684 ranges: Array<[number, number]>,
685 overflowSum: number,
686 penaltySum: number,
687 ) {
688 const remainingParts = actualParts - partIndex;
689 const remainingSentences = sentenceCount - startSentence;
690 if (remainingSentences < remainingParts) return;
691
692 if (partIndex === actualParts - 1) {
693 const score = partScore(startSentence, sentenceCount);
694 const totalOverflow = overflowSum + score.overflow;
695 const totalPenalty = penaltySum + score.deviation + score.overflow * 1000;
696 if (
697 totalOverflow < bestOverflow ||
698 (totalOverflow === bestOverflow && totalPenalty < bestPenalty)
699 ) {
700 bestOverflow = totalOverflow;
701 bestPenalty = totalPenalty;
702 bestRanges = [...ranges, [startSentence, sentenceCount]];
703 }
704 return;
705 }
706
707 const minEnd = startSentence + 1;
708 const maxEnd = sentenceCount - (remainingParts - 1);
709
710 for (let end = minEnd; end <= maxEnd; end += 1) {
711 const score = partScore(startSentence, end);
712 const nextOverflow = overflowSum + score.overflow;
713 const nextPenalty = penaltySum + score.deviation + score.overflow * 1000;
714
715 if (nextOverflow > bestOverflow) continue;
716 if (nextOverflow === bestOverflow && nextPenalty >= bestPenalty) continue;
717
718 ranges.push([startSentence, end]);
719 dfs(partIndex + 1, end, ranges, nextOverflow, nextPenalty);
720 ranges.pop();
721 }
722 }
723
724 dfs(0, 0, [], 0, 0);
725
726 if (!bestRanges || bestRanges.length === 0) {
727 return splitChunkIntoFixedParts(s, actualParts);
728 }
729
730 return bestRanges.map(([startSentence, endSentence]) => {
731 const startIndex = sentenceChunks[startSentence].startIndex;
732 const endIndex = sentenceChunks[endSentence - 1].endIndex;
733 return {
734 text: s.slice(startIndex, endIndex),
735 startIndex,
736 endIndex,
737 };
738 });
739}

1642 if (!Number.isInteger(review.id)) { invalidIds.push(review.id); continue; }
1643 if (!targetIdSet.has(review.id)) extraIds.push(review.id);

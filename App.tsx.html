735function calculateAgeFromBirthday(birthday: string) {
736 const normalized = normalizeBirthdayInput(birthday);
737 if (!normalized) return null;
738 const match = normalized.match(/^(\d{4})(?:-(\d{1,2}))?(?:-(\d{1,2}))?$/);
739 if (!match) return null;
740
741 const [, yearText, monthText, dayText] = match;
742 const year = Number(yearText);
743 const month = monthText ? Number(monthText) : null;
744 const day = dayText ? Number(dayText) : null;
745 const now = new Date();
746
747 if (!Number.isInteger(year) || year < 1900 || year > now.getFullYear()) return null;
748 if (month !== null && (!Number.isInteger(month) || month < 1 || month > 12)) return null;
749 if (day !== null) {
750 if (month === null) return null;
751 const daysInMonth = new Date(year, month, 0).getDate();
752 if (!Number.isInteger(day) || day < 1 || day > daysInMonth) return null;
753 }
754
755 let age = now.getFullYear() - year;
756 const currentMonth = now.getMonth() + 1;
757 const currentDay = now.getDate();
758
759 if (month !== null) {
760 if (currentMonth < month) {
761 age -= 1;
762 } else if (currentMonth === month && day !== null && currentDay < day) {
763 age -= 1;
764 }
765 }
766
767 return age >= 0 ? age : null;
768}
